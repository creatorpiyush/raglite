<TabItem label="TypeScript">

```ts
import { Document } from "raglite-toolkit";

const doc = new Document("./policy.pdf", { llm: { provider: "anthropic" } });
await doc.build();

const answer = await doc.ask("What is the refund policy?");
console.log(answer.text, answer.citations);
```

</TabItem>
<TabItem label="Python">

```python
from raglite import Document

doc = Document("./policy.pdf", {"llm": {"provider": "anthropic"}})
doc.build()

answer = doc.ask("What is the refund policy?")
print(answer.text, answer.citations)
```
